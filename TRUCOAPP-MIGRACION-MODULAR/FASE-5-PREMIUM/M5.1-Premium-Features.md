# 💎 M5.1 - SISTEMA PREMIUM
## Arquitectura: Backend Go + Frontend React

## 📋 OBJETIVO
Implementar un sistema premium completo que monetice la aplicación a través de features avanzadas y experiencias exclusivas, utilizando un backend en Go para gestión de suscripciones y un frontend React para experiencias premium.

---

## 🎯 ALCANCE

### ✅ Lo que INCLUYE
- **Backend**: API de suscripciones en Go con stripe/payment processing
- **Frontend**: UI React para premium con experiencias exclusivas
- **Modelo Freemium**: Free vs Premium ($3/mes) según UX flow original
- **Payment integration**: Stripe, Apple Pay, Google Pay
- **Premium features**: Historial ilimitado, todos los logros, stats avanzadas
- **Limitaciones Free**: 30 días historial, 3 rivalidades, 5 logros máximo
- **Mock Services**: Para desarrollo frontend independiente

### ❌ Lo que NO incluye
- Procesamiento de refunds (soporte manual)
- Integración con facturación empresarial
- Criptomonedas como medio de pago
- Sistema de gifting entre usuarios

```sql
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,

    current_period_start TIMESTAMP NOT NULL,

    -- Metadata
    cancellation_reason TEXT,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW(),

    UNIQUE(user_id) WHERE status = 'active'
```

## 💳 PAYMENT INTEGRATION

### Stripe Integration Service
```go
// internal/payment/stripe_service.go
package payment

import (
    "github.com/stripe/stripe-go/v74"
    "github.com/stripe/stripe-go/v74/customer"
    "github.com/stripe/stripe-go/v74/sub"
    "github.com/stripe/stripe-go/v74/paymentintent"
)

type StripeService struct {
    config *Config
}

type SubscriptionRequest struct {
    UserID       string `json:"user_id"`
    PlanID       string `json:"plan_id"`
    BillingCycle string `json:"billing_cycle"`
    PaymentMethodID string `json:"payment_method_id"`
    TrialDays    int    `json:"trial_days"`
}

func (s *StripeService) CreateSubscription(req SubscriptionRequest) (*SubscriptionResponse, error) {
    // 1. Get or create Stripe customer
    customer, err := s.getOrCreateCustomer(req.UserID)
    if err != nil {
        return nil, err
    }
    
    // 2. Attach payment method
    if req.PaymentMethodID != "" {
        err = s.attachPaymentMethod(customer.ID, req.PaymentMethodID)
        if err != nil {
            return nil, err
        }
    }
    
    // 3. Get plan pricing
    plan, err := s.getPlan(req.PlanID)
    if err != nil {
        return nil, err
    }
    
    stripePriceID := plan.StripePriceIDMonthly
    if req.BillingCycle == "yearly" {
        stripePriceID = plan.StripePriceIDYearly
    }
    
    // 4. Create subscription
    params := &stripe.SubscriptionParams{
        Customer: stripe.String(customer.ID),
        Items: []*stripe.SubscriptionItemsParams{
            {
                Price: stripe.String(stripePriceID),
            },
        },
        PaymentBehavior: stripe.String("default_incomplete"),
        PaymentSettings: &stripe.SubscriptionPaymentSettingsParams{
            SaveDefaultPaymentMethod: stripe.String("on_subscription"),
        },
        Expand: []*string{
            stripe.String("latest_invoice.payment_intent"),
        },
    }
    
    // Add trial if specified
    if req.TrialDays > 0 {
        params.TrialPeriodDays = stripe.Int64(int64(req.TrialDays))
    }
    
    subscription, err := sub.New(params)
    if err != nil {
        return nil, err
    }
    
    // 5. Save to database
    err = s.saveSubscriptionToDB(req.UserID, subscription, plan)
    if err != nil {
        return nil, err
    }
    
    return &SubscriptionResponse{
        Subscription: subscription,
        RequiresAction: subscription.LatestInvoice.PaymentIntent.Status == "requires_action",
        ClientSecret: subscription.LatestInvoice.PaymentIntent.ClientSecret,
    }, nil
}

func (s *StripeService) HandleWebhook(payload []byte, signature string) error {
    event, err := webhook.ConstructEvent(payload, signature, s.config.WebhookSecret)
    if err != nil {
        return err
    }
    
    switch event.Type {
    case "invoice.payment_succeeded":
        err = s.handlePaymentSucceeded(event.Data.Object)
    case "invoice.payment_failed":
        err = s.handlePaymentFailed(event.Data.Object)
    case "customer.subscription.updated":
        err = s.handleSubscriptionUpdated(event.Data.Object)
    case "customer.subscription.deleted":
        err = s.handleSubscriptionDeleted(event.Data.Object)
    default:
        log.Printf("Unhandled event type: %s", event.Type)
    }
    
    return err
}
```

## 📱 FRONTEND: REACT HOOKS

### usePremium Hook
```javascript
// hooks/usePremium.js
import { useState, useEffect, useCallback, useMemo } from 'react';
import { useAuth } from './useAuth';
import { premiumAPI } from '../services/premiumAPI';
import { mockPremiumService } from '../services/mockPremiumService';

export const usePremium = () => {
  const { isAuthenticated, user } = useAuth();
  const [subscription, setSubscription] = useState(null);
  const [plans, setPlans] = useState([]);
  const [features, setFeatures] = useState({});
  const [usage, setUsage] = useState({});
  const [isLoading, setIsLoading] = useState(false);
  const [error, setError] = useState(null);
  
  // Usar mock o API real según configuración
  const api = process.env.REACT_APP_USE_MOCK_PREMIUM === 'true' 
    ? mockPremiumService 
    : premiumAPI;
  
  const loadSubscription = useCallback(async () => {
    if (!isAuthenticated) return;
    
    try {
      setIsLoading(true);
      const response = await api.getSubscription();
      setSubscription(response.subscription);
      setUsage(response.usage || {});
    } catch (err) {
      if (err.status !== 404) { // 404 = no subscription
        setError(err.message);
      }
    } finally {
      setIsLoading(false);
    }
  }, [isAuthenticated, api]);
  
  const loadPlans = useCallback(async () => {
    try {
      const response = await api.getPlans();
      setPlans(response.plans);
    } catch (err) {
      setError(err.message);
    }
  }, [api]);
  
  const loadFeatures = useCallback(async () => {
    if (!isAuthenticated) return;
    
    try {
      const response = await api.getFeatures();
      setFeatures(response.features);
      if (response.usage_status) {
        setUsage(response.usage_status);
      }
    } catch (err) {
      setError(err.message);
    }
  }, [isAuthenticated, api]);
  
  const subscribe = useCallback(async (planId, billingCycle, paymentMethodId, trialDays) => {
    try {
      setIsLoading(true);
      const response = await api.subscribe({
        plan_id: planId,
        billing_cycle: billingCycle,
        payment_method_id: paymentMethodId,
        trial_days: trialDays
      });
      
      setSubscription(response.subscription);
      await loadFeatures(); // Refresh features
      
      return response;
    } catch (err) {
      setError(err.message);
      throw err;
    } finally {
      setIsLoading(false);
    }
  }, [api, loadFeatures]);
  
  const startTrial = useCallback(async (planId, trialDays = 7) => {
    try {
      setIsLoading(true);
      const response = await api.startTrial(planId, trialDays);
      setSubscription(response.trial);
      await loadFeatures();
      return response;
    } catch (err) {
      setError(err.message);
      throw err;
    } finally {
      setIsLoading(false);
    }
  }, [api, loadFeatures]);
  
  const cancelSubscription = useCallback(async (immediately = false) => {
    try {
      setIsLoading(true);
      await api.updateSubscription({
        action: 'cancel',
        cancel_immediately: immediately
      });
      
      await loadSubscription(); // Refresh subscription
      await loadFeatures(); // Refresh features
      
      return true;
    } catch (err) {
      setError(err.message);
      return false;
    } finally {
      setIsLoading(false);
    }
  }, [api, loadSubscription, loadFeatures]);
  
  const changePlan = useCallback(async (newPlanId) => {
    try {
      setIsLoading(true);
      await api.updateSubscription({
        action: 'change_plan',
        plan_id: newPlanId
      });
      
      await loadSubscription();
      await loadFeatures();
      
      return true;
    } catch (err) {
      setError(err.message);
      return false;
    } finally {
      setIsLoading(false);
    }
  }, [api, loadSubscription, loadFeatures]);
  
  // Feature gates
  const hasFeature = useCallback((featureName) => {
    return features[featureName] === true;
  }, [features]);
  
  const canUseFeature = useCallback((featureName, usageKey) => {
    const hasAccess = hasFeature(featureName);
    
    if (!hasAccess) return false;
    if (!usageKey || !usage[usageKey]) return true;
    
    const usageInfo = usage[usageKey];
    return usageInfo.unlimited || usageInfo.used < usageInfo.limit;
  }, [hasFeature, usage]);
  
  const getUsagePercentage = useCallback((usageKey) => {
    if (!usage[usageKey]) return 0;
    const usageInfo = usage[usageKey];
    return usageInfo.unlimited ? 0 : usageInfo.percentage || 0;
  }, [usage]);
  
  useEffect(() => {
    loadPlans();
    if (isAuthenticated) {
      loadSubscription();
      loadFeatures();
    }
  }, [isAuthenticated, loadPlans, loadSubscription, loadFeatures]);
  
  // Computed values
  const currentPlan = useMemo(() => {
    if (!subscription) return plans.find(p => p.name === 'free');
    return plans.find(p => p.id === subscription.plan_id) || 
           plans.find(p => p.name === 'free');
  }, [subscription, plans]);
  
  const isFreePlan = useMemo(() => {
    return !subscription || currentPlan?.name === 'free';
  }, [subscription, currentPlan]);
  
  const isOnTrial = useMemo(() => {
    if (!subscription?.trial_end) return false;
    return new Date(subscription.trial_end) > new Date();
  }, [subscription]);
  
  const trialDaysRemaining = useMemo(() => {
    if (!isOnTrial) return 0;
    const trialEnd = new Date(subscription.trial_end);
    const now = new Date();
    const diffTime = trialEnd - now;
    return Math.ceil(diffTime / (1000 * 60 * 60 * 24));
  }, [isOnTrial, subscription]);
  
  return {
    // Estado
    subscription,
    plans,
    features,
    usage,
    currentPlan,
    isLoading,
    error,
    
    // Estado computado
    isFreePlan,
    isOnTrial,
    trialDaysRemaining,
    
    // Acciones
    subscribe,
    startTrial,
    cancelSubscription,
    changePlan,
    
    // Feature gates
    hasFeature,
    canUseFeature,
    getUsagePercentage,
    
    // Utilidades
    refreshSubscription: loadSubscription,
    refreshFeatures: loadFeatures
  };
};
```

### usePaymentFlow Hook
```javascript
// hooks/usePaymentFlow.js
import { useState, useCallback } from 'react';
import { loadStripe } from '@stripe/stripe-js';
import { usePremium } from './usePremium';

const stripePromise = loadStripe(process.env.REACT_APP_STRIPE_PUBLISHABLE_KEY);

export const usePaymentFlow = () => {
  const { subscribe, startTrial } = usePremium();
  const [isProcessing, setIsProcessing] = useState(false);
  const [paymentError, setPaymentError] = useState(null);
  
  const processPayment = useCallback(async (planId, billingCycle, paymentMethod) => {
    try {
      setIsProcessing(true);
      setPaymentError(null);
      
      const response = await subscribe(planId, billingCycle, paymentMethod.id);
      
      if (response.requires_action) {
        const stripe = await stripePromise;
        const { error } = await stripe.confirmCardPayment(response.client_secret);
        
        if (error) {
          setPaymentError(error.message);
          return { success: false, error: error.message };
        }
      }
      
      return { success: true, subscription: response.subscription };
      
    } catch (error) {
      setPaymentError(error.message);
      return { success: false, error: error.message };
    } finally {
      setIsProcessing(false);
    }
  }, [subscribe]);
  
  const startFreeTrial = useCallback(async (planId, trialDays = 7) => {
    try {
      setIsProcessing(true);
      setPaymentError(null);
      
      const response = await startTrial(planId, trialDays);
      return { success: true, trial: response.trial };
      
    } catch (error) {
      setPaymentError(error.message);
      return { success: false, error: error.message };
    } finally {
      setIsProcessing(false);
    }
  }, [startTrial]);
  
  return {
    isProcessing,
    paymentError,
    processPayment,
    startFreeTrial,
    clearError: () => setPaymentError(null)
  };
};
```
