# 👥 M3.1 - SISTEMA SOCIAL COMPLETO
## Arquitectura: Backend Go + Frontend React

## 📋 OBJETIVO
Implementar el sistema social completo de TrucoApp que incluye amigos, rankings locales/regionales, sistema de desafíos y activity feed, utilizando un backend en Go para gestión social y un frontend React según el UX flow original.

---

## 🎯 ALCANCE

### ✅ Lo que INCLUYE
- **Sistema de Amigos**: Búsqueda, invitaciones, gestión de amigos
- **Rankings Locales**: Córdoba, Buenos Aires, Rosario, etc. por ciudad
- **Rankings Regionales**: Por provincia/región
- **Rankings Nacionales**: Ranking Argentina completo
- **Sistema de Desafíos**: Envío, aceptación, rechazos con contexto
- **Activity Feed**: Actividad de amigos y social feed
- **Progreso al Top**: "Al ritmo actual: 3-4 meses" para subir ranking
- **Backend Go**: PostgreSQL + Redis para rankings y relaciones
- **Mock Services**: Para desarrollo frontend independiente

### ❌ Lo que NO incluye
- Chat en tiempo real (Fase 4)
- Notificaciones push (Fase 4)
- Rankings internacionales
- Torneos organizados (futuro)

## 🗄️ BACKEND: DATABASE SCHEMA

### Friendships Table
```sql
CREATE TABLE friendships (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    friend_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    status VARCHAR(20) NOT NULL DEFAULT 'active', -- 'active', 'blocked'
    created_at TIMESTAMP DEFAULT NOW(),
    
    -- Prevent duplicate friendships
    UNIQUE(user_id, friend_id),
    
    -- Prevent self-friending
    CHECK (user_id != friend_id)
);

CREATE INDEX idx_friendships_user_id ON friendships(user_id);
CREATE INDEX idx_friendships_friend_id ON friendships(friend_id);
CREATE INDEX idx_friendships_status ON friendships(status);
```

### Friend Requests Table
```sql
CREATE TABLE friend_requests (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    from_user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    to_user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    status VARCHAR(20) NOT NULL DEFAULT 'pending', -- 'pending', 'accepted', 'rejected', 'cancelled'
    message TEXT,
    created_at TIMESTAMP DEFAULT NOW(),
    responded_at TIMESTAMP,
    
    -- Prevent duplicate requests
    UNIQUE(from_user_id, to_user_id),
    
    -- Prevent self-requests
    CHECK (from_user_id != to_user_id)
);

CREATE INDEX idx_friend_requests_from_user ON friend_requests(from_user_id);
CREATE INDEX idx_friend_requests_to_user ON friend_requests(to_user_id);
CREATE INDEX idx_friend_requests_status ON friend_requests(status);
CREATE INDEX idx_friend_requests_created_at ON friend_requests(created_at);
```

### User Blocks Table
```sql
CREATE TABLE user_blocks (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    blocker_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    blocked_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    reason VARCHAR(100),
    created_at TIMESTAMP DEFAULT NOW(),
    
    -- Prevent duplicate blocks
    UNIQUE(blocker_id, blocked_id),
    
    -- Prevent self-blocking
    CHECK (blocker_id != blocked_id)
);

CREATE INDEX idx_user_blocks_blocker ON user_blocks(blocker_id);
CREATE INDEX idx_user_blocks_blocked ON user_blocks(blocked_id);
```

### Social Activities Table
```sql
CREATE TABLE social_activities (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    activity_type VARCHAR(50) NOT NULL, -- 'game_won', 'streak_started', 'achievement_unlocked'
    activity_data JSONB,
    visibility VARCHAR(20) DEFAULT 'friends', -- 'public', 'friends', 'private'
    created_at TIMESTAMP DEFAULT NOW(),
    
    -- For efficient friend feed queries
    friend_feed_eligible BOOLEAN DEFAULT TRUE
);

CREATE INDEX idx_social_activities_user_id ON social_activities(user_id);
CREATE INDEX idx_social_activities_created_at ON social_activities(created_at);
CREATE INDEX idx_social_activities_type ON social_activities(activity_type);
CREATE INDEX idx_social_activities_visibility ON social_activities(visibility);
CREATE INDEX idx_social_activities_feed ON social_activities(friend_feed_eligible, created_at) 
    WHERE friend_feed_eligible = TRUE;
```

### Rankings Table
```sql
CREATE TABLE rankings (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    
    -- Ranking type and scope
    ranking_type VARCHAR(20) NOT NULL, -- 'local', 'regional', 'national'
    location_scope VARCHAR(100) NOT NULL, -- 'Córdoba', 'Buenos Aires', 'Argentina'
    
    -- Position and stats
    current_rank INTEGER NOT NULL,
    previous_rank INTEGER,
    rank_change INTEGER DEFAULT 0, -- positive = up, negative = down
    
    -- Qualification stats (snapshot for ranking period)
    total_games INTEGER NOT NULL,
    total_wins INTEGER NOT NULL,
    win_rate DECIMAL(5,2) NOT NULL,
    ranking_score DECIMAL(10,2) NOT NULL, -- Calculated ranking score
    
    -- Progress metrics
    games_to_next_rank INTEGER,
    wins_needed INTEGER,
    win_rate_needed DECIMAL(5,2),
    estimated_time_months DECIMAL(3,1), -- "Al ritmo actual: 3-4 meses"
    
    -- Ranking period
    period_start DATE NOT NULL,
    period_end DATE NOT NULL,
    is_current BOOLEAN DEFAULT TRUE,
    
    -- Metadata
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW(),
    
    UNIQUE(user_id, ranking_type, location_scope, period_start)
);

CREATE INDEX idx_rankings_current ON rankings(ranking_type, location_scope, is_current, current_rank);
CREATE INDEX idx_rankings_user ON rankings(user_id, ranking_type);
CREATE INDEX idx_rankings_score ON rankings(ranking_type, location_scope, ranking_score DESC);
```

### Challenges Table
```sql
CREATE TABLE challenges (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    challenger_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    challenged_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    
    -- Challenge details
    challenge_type VARCHAR(20) DEFAULT 'revenge', -- 'revenge', 'friendly', 'ranking'
    message TEXT,
    
    -- Context from last game/rivalry
    last_game_result VARCHAR(10), -- 'won', 'lost' by challenger
    last_game_score VARCHAR(10), -- "30-18"
    head_to_head_record VARCHAR(10), -- "14-8" 
    head_to_head_winner VARCHAR(20), -- "challenger" or "challenged"
    context_message TEXT, -- "Lo dormiste afuera", "Te ganó 30-24 ayer"
    
    -- Challenge status
    status VARCHAR(20) DEFAULT 'pending', -- 'pending', 'accepted', 'rejected', 'expired', 'completed'
    response_message TEXT,
    
    -- Game reference (if completed)
    game_id UUID REFERENCES games(id),
    
    -- Timestamps
    created_at TIMESTAMP DEFAULT NOW(),
    responded_at TIMESTAMP,
    expires_at TIMESTAMP DEFAULT (NOW() + INTERVAL '7 days'),
    completed_at TIMESTAMP
);

CREATE INDEX idx_challenges_challenged ON challenges(challenged_id, status);
CREATE INDEX idx_challenges_challenger ON challenges(challenger_id);
CREATE INDEX idx_challenges_status ON challenges(status);
CREATE INDEX idx_challenges_expires ON challenges(expires_at) WHERE status = 'pending';
```

```javascript
  const loadFriends = async () => {
    if (!isAuthenticated) return;
    
    try {
      setIsLoading(true);
      const response = await api.getFriends();
      setFriends(response.friends);
    } catch (err) {
      setError(err.message);
    } finally {
      setIsLoading(false);
    }
  };
  
  const loadFriendRequests = async () => {
    if (!isAuthenticated) return;
    
    try {
      const response = await api.getFriendRequests();
      setFriendRequests(response.requests);
    } catch (err) {
      setError(err.message);
    }
  };
  
  const sendFriendRequest = async (userId, message = '') => {
    try {
      setIsLoading(true);
      await api.sendFriendRequest(userId, message);
      await loadFriendRequests(); // Refresh requests
      return true;
    } catch (err) {
      setError(err.message);
      return false;
    } finally {
      setIsLoading(false);
    }
  };
  
  const respondToRequest = async (requestId, action, message = '') => {
    try {
      setIsLoading(true);
      await api.respondToFriendRequest(requestId, action, message);
      await loadFriendRequests();
      
      if (action === 'accept') {
        await loadFriends(); // Refresh friends list
      }
      
      return true;
    } catch (err) {
      setError(err.message);
      return false;
    } finally {
      setIsLoading(false);
    }
  };
  
  const removeFriend = async (friendId) => {
    try {
      setIsLoading(true);
      await api.removeFriend(friendId);
      setFriends(friends.filter(f => f.id !== friendId));
      return true;
    } catch (err) {
      setError(err.message);
      return false;
    } finally {
      setIsLoading(false);
    }
  };
  
  const blockUser = async (userId, reason = '') => {
    try {
      setIsLoading(true);
      await api.blockUser(userId, reason);
      setFriends(friends.filter(f => f.id !== userId));
      return true;
    } catch (err) {
      setError(err.message);
      return false;
    } finally {
      setIsLoading(false);
    }
  };
  
  useEffect(() => {
    if (isAuthenticated) {
      loadFriends();
      loadFriendRequests();
    }
  }, [isAuthenticated]);
  
  return {
    // Estado
    friends,
    friendRequests,
    isLoading,
    error,
    
    // Acciones
    sendFriendRequest,
    respondToRequest,
    removeFriend,
    blockUser,
    
    // Utilidades
    refreshFriends: loadFriends,
    refreshRequests: loadFriendRequests,
    pendingRequestsCount: friendRequests.filter(r => r.type === 'received').length,
    onlineFriendsCount: friends.filter(f => f.is_online).length
  };
};
```

```javascript
// hooks/useUserSearch.js
import { useState, useCallback, useMemo } from 'react';
import { useAuth } from './useAuth';
import { socialAPI } from '../services/socialAPI';
import { mockSocialService } from '../services/mockSocialService';
import { useDebounce } from './useDebounce';

export const useUserSearch = () => {
  const { isAuthenticated } = useAuth();
  const [searchTerm, setSearchTerm] = useState('');
  const [searchResults, setSearchResults] = useState([]);
  const [suggestions, setSuggestions] = useState([]);
  const [isLoading, setIsLoading] = useState(false);
  const [error, setError] = useState(null);
  
  const debouncedSearchTerm = useDebounce(searchTerm, 300);
  
  // Usar mock o API real según configuración
  const api = process.env.REACT_APP_USE_MOCK_SOCIAL === 'true' 
    ? mockSocialService 
    : socialAPI;
  
  const searchUsers = useCallback(async (term) => {
    if (!isAuthenticated || !term.trim()) {
      setSearchResults([]);
      return;
    }
    
    try {
      setIsLoading(true);
      const response = await api.searchUsers(term);
      setSearchResults(response.users);
    } catch (err) {
      setError(err.message);
      setSearchResults([]);
    } finally {
      setIsLoading(false);
    }
  }, [isAuthenticated, api]);
  
  const loadSuggestions = useCallback(async () => {
    if (!isAuthenticated) return;
    
    try {
      const response = await api.getFriendSuggestions();
      setSuggestions(response.suggestions);
    } catch (err) {
      setError(err.message);
    }
  }, [isAuthenticated, api]);
  
  // Buscar cuando el término cambie
  useEffect(() => {
    searchUsers(debouncedSearchTerm);
  }, [debouncedSearchTerm, searchUsers]);
  
  // Cargar sugerencias al montar
  useEffect(() => {
    loadSuggestions();
  }, [loadSuggestions]);
  
  const filteredResults = useMemo(() => {
    return searchResults.filter(user => 
      user.relationship_status !== 'blocked'
    );
  }, [searchResults]);
  
  return {
    // Estado de búsqueda
    searchTerm,
    setSearchTerm,
    searchResults: filteredResults,
    suggestions,
    isLoading,
    error,
    
    // Utilidades
    hasSearchTerm: searchTerm.length > 0,
    hasResults: filteredResults.length > 0,
    refreshSuggestions: loadSuggestions,
    clearSearch: () => {
      setSearchTerm('');
      setSearchResults([]);
      setError(null);
    }
  };
};
```
