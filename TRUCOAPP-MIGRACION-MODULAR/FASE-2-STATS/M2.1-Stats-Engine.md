# 📊 M2.1 - MOTOR DE ESTADÍSTICAS
## Arquitectura: Backend Go + Frontend React

## 📋 OBJETIVO
Desarrollar un sistema robusto de cálculo y tracking de estadísticas que transforme los datos de juegos en insights significativos y adictivos para los usuarios, utilizando un backend en Go para cálculos pesados y un frontend React para visualización.

---

## 🎯 ALCANCE

### ✅ Lo que INCLUYE
- **Backend**: Motor de cálculo de estadísticas en Go con PostgreSQL
- **Frontend**: Hooks React para consumo de stats con cache local
- **Estadísticas Individuales**: 
  - Básicas: Partidos, victorias, WR%, rachas
  - Avanzadas: Promedio en derrotas, promedio rival, dormidas
  - Temporales: Por día de semana, por horario, progresión mensual
- **Estadísticas de Rivalidades**: Head-to-head detallado con historial
- **Estadísticas de Equipos**: Duplas y tríos con análisis de sinergia
- **Sistema de agregaciones**: Diarias, semanales, mensuales, anuales
- **Cache inteligente**: Redis + cache local para optimización
- **Mock Services**: Para desarrollo frontend independiente

### ❌ Lo que NO incluye
- UI de estadísticas (M2.2)
- Rankings globales públicos (M3.2)
- Sistema de logros (M4.1)
- Analytics con ML (futuro)

```go
// Obtener stats de usuario con cache
func (s *Service) GetUserStats(ctx context.Context, userID string) (*UserStats, error) {
    // 1. Intentar cache de Redis
    cacheKey := fmt.Sprintf("user_stats:%s", userID)
    cached, err := s.redis.Get(ctx, cacheKey).Result()
    if err == nil {
        var stats UserStats
        if json.Unmarshal([]byte(cached), &stats) == nil {
            return &stats, nil
        }
    }
    
    // 2. Cargar desde PostgreSQL
    stats, err := s.loadUserStatsFromDB(ctx, userID)
    if err != nil {
        return nil, fmt.Errorf("failed to load user stats: %w", err)
    }
    
    // 3. Cache en Redis por 5 minutos
    statsJSON, _ := json.Marshal(stats)
    s.redis.Set(ctx, cacheKey, statsJSON, 5*time.Minute)
    
    return stats, nil
}

// Actualizar stats después de un juego
func (s *Service) UpdateStatsAfterGame(ctx context.Context, gameID string) error {
    // 1. Obtener datos del juego
    game, err := s.getGameData(ctx, gameID)
    if err != nil {
        return fmt.Errorf("failed to get game data: %w", err)
    }
    
    // 2. Calcular stats del juego
    gameStats := s.calculator.CalculateGameStats(game)
    
    // 3. Iniciar transacción
    tx, err := s.db.BeginTx(ctx, nil)
    if err != nil {
        return err
    }
    defer tx.Rollback()
    
    // 4. Insertar game_stats
    if err := s.insertGameStats(ctx, tx, gameStats); err != nil {
        return err
    }
    
    // 5. Actualizar user_stats
    if err := s.updateUserStats(ctx, tx, game.UserID, gameStats); err != nil {
        return err
    }
    
    // 6. Actualizar rivalries si aplica
    if game.OpponentID != nil {
        if err := s.updateRivalryStats(ctx, tx, game.UserID, *game.OpponentID, gameStats); err != nil {
            return err
        }
    }
    
    // 7. Commit transacción
    if err := tx.Commit(); err != nil {
        return err
    }
    
    // 8. Invalidar cache
    s.redis.Del(ctx, fmt.Sprintf("user_stats:%s", game.UserID))
    if game.OpponentID != nil {
        s.redis.Del(ctx, fmt.Sprintf("user_stats:%s", *game.OpponentID))
        s.redis.Del(ctx, fmt.Sprintf("rivalry:%s:%s", game.UserID, *game.OpponentID))
    }
    
    // 9. Actualizar agregaciones en background
    go s.aggregator.UpdateAggregations(context.Background(), game.UserID, gameStats)
    
    return nil
}
```

### Stats Calculator
```go
// internal/stats/calculator.go
package stats

type Calculator struct{}

type GameData struct {
    ID           string     `json:"id"`
    UserID       string     `json:"user_id"`
    OpponentID   *string    `json:"opponent_id"`
    UserPoints   int        `json:"user_points"`
    OpponentPoints int      `json:"opponent_points"`
    TotalPoints  int        `json:"total_points"`
    StartedAt    time.Time  `json:"started_at"`
    FinishedAt   time.Time  `json:"finished_at"`
    DurationMs   int64      `json:"duration_ms"`
    Moves        []GameMove `json:"moves"`
}

type GameMove struct {
    ID             string    `json:"id"`
    Action         string    `json:"action"` // '+', '-', 'falta_envido'
    Team           string    `json:"team"`   // 'nos', 'ellos'
    PreviousPoints int       `json:"previous_points"`
    NewPoints      int       `json:"new_points"`
    Timestamp      time.Time `json:"timestamp"`
}

type GameStats struct {
    GameID         string    `json:"game_id"`
    UserID         string    `json:"user_id"`
    IsWin          bool      `json:"is_win"`
    PointsScored   int       `json:"points_scored"`
    PointsConceded int       `json:"points_conceded"`
    PointDifference int      `json:"point_difference"`
    TotalPoints    int       `json:"total_points"`
    DurationMs     int64     `json:"duration_ms"`
    
    // Characteristics
    IsShutoutGiven    bool `json:"is_shutout_given"`
    IsShutoutReceived bool `json:"is_shutout_received"`
    IsCloseGame       bool `json:"is_close_game"`
    IsBlowout         bool `json:"is_blowout"`
    IsComeback        bool `json:"is_comeback"`
    IsChoke          bool `json:"is_choke"`
    
    // Performance
    MaxLead           int `json:"max_lead"`
    MaxDeficit        int `json:"max_deficit"`
    LeadChanges       int `json:"lead_changes"`
    PressureSituations int `json:"pressure_situations"`
    PressureWins      int `json:"pressure_wins"`
    
    // Temporal
    PlayedAt    time.Time `json:"played_at"`
    DayOfWeek   int       `json:"day_of_week"`
    HourOfDay   int       `json:"hour_of_day"`
}

func (c *Calculator) CalculateGameStats(game *GameData) *GameStats {
    stats := &GameStats{
        GameID:          game.ID,
        UserID:          game.UserID,
        IsWin:           game.UserPoints > game.OpponentPoints,
        PointsScored:    game.UserPoints,
        PointsConceded:  game.OpponentPoints,
        PointDifference: game.UserPoints - game.OpponentPoints,
        TotalPoints:     game.TotalPoints,
        DurationMs:      game.DurationMs,
        PlayedAt:        game.FinishedAt,
        DayOfWeek:       int(game.FinishedAt.Weekday()),
        HourOfDay:       game.FinishedAt.Hour(),
    }
    
    // Game characteristics
    stats.IsShutoutGiven = stats.IsWin && game.OpponentPoints < 15
    stats.IsShutoutReceived = !stats.IsWin && game.UserPoints < 15
    stats.IsCloseGame = abs(stats.PointDifference) <= 5
    stats.IsBlowout = abs(stats.PointDifference) >= 15
    
    // Analyze game progression
    if len(game.Moves) > 0 {
        progression := c.analyzeGameProgression(game.Moves, game.UserPoints, game.OpponentPoints)
        stats.IsComeback = progression.IsComeback
        stats.IsChoke = progression.IsChoke
        stats.MaxLead = progression.MaxLead
        stats.MaxDeficit = progression.MaxDeficit
        stats.LeadChanges = progression.LeadChanges
        stats.PressureSituations = progression.PressureSituations
        stats.PressureWins = progression.PressureWins
    }
    
    return stats
}

type GameProgression struct {
    IsComeback         bool
    IsChoke           bool
    MaxLead           int
    MaxDeficit        int
    LeadChanges       int
    PressureSituations int
    PressureWins      int
}

func (c *Calculator) analyzeGameProgression(moves []GameMove, finalUserPoints, finalOpponentPoints int) *GameProgression {
    prog := &GameProgression{}
    
    userPoints := 0
    opponentPoints := 0
    lastLead := 0 // positive = user leads, negative = opponent leads
    maxUserLead := 0
    maxOpponentLead := 0
    userWasBehindBy10Plus := false
    userWasAheadBy10Plus := false
    
    for _, move := range moves {
        // Update points based on move
        if move.Team == "nos" { // user team
            if move.Action == "+" {
                userPoints++
            } else if move.Action == "-" {
                userPoints--
            }
        } else { // opponent team
            if move.Action == "+" {
                opponentPoints++
            } else if move.Action == "-" {
                opponentPoints--
            }
        }
        
        currentLead := userPoints - opponentPoints
        
        // Track max leads
        if currentLead > maxUserLead {
            maxUserLead = currentLead
        }
        if currentLead < -maxOpponentLead {
            maxOpponentLead = -currentLead
        }
        
        // Track lead changes
        if (lastLead > 0 && currentLead < 0) || (lastLead < 0 && currentLead > 0) {
            prog.LeadChanges++
        }
        
        // Track deficit/lead patterns
        if currentLead <= -10 {
            userWasBehindBy10Plus = true
        }
        if currentLead >= 10 {
            userWasAheadBy10Plus = true
        }
        
        // Pressure situations (opponent at 29 points)
        if opponentPoints == 29 {
            prog.PressureSituations++
            if finalUserPoints == 30 && finalUserPoints > finalOpponentPoints {
                prog.PressureWins++
            }
        }
        
        lastLead = currentLead
    }
    
    prog.MaxLead = maxUserLead
    prog.MaxDeficit = maxOpponentLead
    
    // Determine comeback/choke
    userWon := finalUserPoints > finalOpponentPoints
    prog.IsComeback = userWasBehindBy10Plus && userWon
    prog.IsChoke = userWasAheadBy10Plus && !userWon
    
    return prog
}

func abs(x int) int {
    if x < 0 {
        return -x
    }
    return x
}
```

```javascript
// hooks/useUserStats.js
import { useState, useEffect, useCallback, useMemo } from 'react';
import { statsAPI } from '../services/statsAPI';
import { mockStatsService } from '../services/mockStatsService';

export const useUserStats = (userId, options = {}) => {
  const [stats, setStats] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  
  const { 
    autoRefresh = true, 
    refreshInterval = 5 * 60 * 1000, // 5 minutos
    useCache = true 
  } = options;
  
  // Usar mock o API real
  const api = useMemo(() => {
    return process.env.REACT_APP_USE_MOCK_STATS === 'true' 
      ? mockStatsService 
      : statsAPI;
  }, []);
  
  const loadStats = useCallback(async () => {
    if (!userId) return;
    
    try {
      setLoading(true);
      setError(null);
      
      // Verificar cache local primero
      if (useCache) {
        const cached = localStorage.getItem(`stats_${userId}`);
        if (cached) {
          const { data, timestamp } = JSON.parse(cached);
          const age = Date.now() - timestamp;
          
          // Cache válido por 2 minutos
          if (age < 2 * 60 * 1000) {
            setStats(data);
            setLoading(false);
            return;
          }
        }
      }
      
      const userStats = await api.getUserStats(userId);
      setStats(userStats);
      
      // Guardar en cache local
      if (useCache) {
        localStorage.setItem(`stats_${userId}`, JSON.stringify({
          data: userStats,
          timestamp: Date.now()
        }));
      }
      
    } catch (err) {
      setError(err.message);
      console.error('Error loading stats:', err);
    } finally {
      setLoading(false);
    }
  }, [userId, api, useCache]);
  
  // Cargar stats inicial
  useEffect(() => {
    loadStats();
  }, [loadStats]);
  
  // Auto refresh
  useEffect(() => {
    if (!autoRefresh || !userId) return;
    
    const interval = setInterval(loadStats, refreshInterval);
    return () => clearInterval(interval);
  }, [autoRefresh, refreshInterval, loadStats]);
  
  const refreshStats = useCallback(() => {
    // Limpiar cache local
    if (useCache) {
      localStorage.removeItem(`stats_${userId}`);
    }
    loadStats();
  }, [userId, useCache, loadStats]);
  
  const invalidateCache = useCallback(() => {
    if (useCache) {
      localStorage.removeItem(`stats_${userId}`);
    }
  }, [userId, useCache]);
  
  return {
    // Estado principal
    stats,
    loading,
    error,
    
    // Acciones
    refreshStats,
    invalidateCache,
    
    // Shortcuts para stats comunes
    winRate: stats?.win_rate || 0,
    totalGames: stats?.games_played || 0,
    totalWins: stats?.games_won || 0,
    totalLosses: stats?.games_lost || 0,
    currentStreak: stats?.current_streak || 0,
    streakType: stats?.current_streak_type || 'none',
    bestWinStreak: stats?.best_win_streak || 0,
    
    // Stats avanzadas
    comebackGames: stats?.comeback_games || 0,
    chokeGames: stats?.choke_games || 0,
    pressurePerformance: stats?.pressure_performance || 0,
    clutchWins: stats?.clutch_wins || 0,
    
    // Dominancia
    shutoutsGiven: stats?.shutouts_given || 0,
    shutoutsReceived: stats?.shutouts_received || 0,
    closeGames: stats?.close_games || 0,
    blowoutWins: stats?.blowout_wins || 0,
    
    // Utilidades computadas
    hasPlayedGames: (stats?.games_played || 0) > 0,
    isOnWinStreak: stats?.current_streak_type === 'win',
    isOnLossStreak: stats?.current_streak_type === 'loss',
    averagePointsPerGame: stats?.avg_points_per_game || 0,
    dominanceRatio: stats?.shutouts_given / Math.max(stats?.shutouts_received || 1, 1),
    
    // Metadatos
    lastUpdated: stats?.last_updated
  };
};
```
