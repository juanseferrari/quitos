## 🔌 HOOKS PERSONALIZADOS

### Hook Principal: useGame
```javascript
// hooks/useGame.js
export const useGame = () => {
  const { state, dispatch } = useContext(GameContext);
  
  const sumarPunto = useCallback((player) => {
    dispatch(gameActions.addPoint(player));
  }, [dispatch]);
  
  const restarPunto = useCallback((player) => {
    dispatch(gameActions.subtractPoint(player));
  }, [dispatch]);
  
  const nuevoPartido = useCallback((config = {}) => {
    dispatch(gameActions.newGame(config));
  }, [dispatch]);
  
  const restaurarPartida = useCallback((savedGame) => {
    dispatch(gameActions.restoreGame(savedGame));
  }, [dispatch]);
  
  return {
    // Estado
    game: state.game,
    config: state.config,
    meta: state.meta,
    
    // Acciones
    sumarPunto,
    restarPunto,
    nuevoPartido,
    restaurarPartida,
    
    // Utilidades
    hayGanador: state.game.ganador !== null,
    partidaEnProgreso: state.game.puntosNos > 0 || state.game.puntosEllos > 0,
    needsSync: state.meta.needsSync
  };
};
```

### Hook de Persistencia: useGamePersistence
```javascript
// hooks/useGamePersistence.js
export const useGamePersistence = () => {
  const { state, dispatch } = useContext(GameContext);
  
  // Auto-save cuando cambie el estado
  useEffect(() => {
    if (state.meta.needsSync) {
      saveToLocal(state);
      dispatch({ type: 'SYNC_COMPLETED' });
    }
  }, [state.meta.needsSync]);
  
  const saveToLocal = async (gameState) => {
    try {
      const dataToSave = {
        game: gameState.game,
        localData: gameState.localData,
        timestamp: Date.now()
      };
      
      await AsyncStorage.setItem(
        'rey-del-truco-state', 
        JSON.stringify(dataToSave)
      );
    } catch (error) {
      console.error('Error saving game:', error);
    }
  };
  
  const loadFromLocal = async () => {
    try {
      const saved = await AsyncStorage.getItem('rey-del-truco-state');
      if (saved) {
        const data = JSON.parse(saved);
        
        // Verificar si es menor a 24 horas
        const isRecent = Date.now() - data.timestamp < 24 * 60 * 60 * 1000;
        
        if (isRecent && !data.game.ganador) {
          return data;
        }
      }
    } catch (error) {
      console.error('Error loading game:', error);
    }
    return null;
  };
  
  return {
    saveToLocal,
    loadFromLocal,
    clearLocal: () => AsyncStorage.removeItem('rey-del-truco-state')
  };
};
```

## 🎨 CAMBIOS EN COMPONENTES

### Provider Principal
```javascript
// providers/GameProvider.js
export const GameProvider = ({ children }) => {
  const [state, dispatch] = useReducer(gameReducer, initialState);
  
  // Cargar estado inicial
  useEffect(() => {
    loadInitialState();
  }, []);
  
  const loadInitialState = async () => {
    try {
      const saved = await AsyncStorage.getItem('rey-del-truco-state');
      if (saved) {
        const data = JSON.parse(saved);
        dispatch({ type: 'RESTORE_GAME', payload: data });
      }
    } catch (error) {
      console.error('Error loading initial state:', error);
    }
  };
  
  return (
    <GameContext.Provider value={{ state, dispatch }}>
      {children}
    </GameContext.Provider>
  );
};
```

describe('gameReducer', () => {
  test('ADD_POINT aumenta puntos correctamente', () => {
    const state = initialState;
    const action = { type: 'ADD_POINT', payload: { player: 'puntosNos' } };

      ...initialState,
      game: { ...initialState.game, puntosNos: 15, puntosEllos: 12 }

    const action = { type: 'NEW_GAME', payload: {} };
    const newState = gameReducer(state, action);
    
    expect(newState.game.puntosNos).toBe(0);
    expect(newState.game.puntosEllos).toBe(0);
    expect(newState.game.id).toBeTruthy();
  });
});
