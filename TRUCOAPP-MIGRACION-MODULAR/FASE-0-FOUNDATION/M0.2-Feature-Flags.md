```javascript
// providers/FeatureFlagsProvider.js
import React, { createContext, useContext, useReducer, useEffect } from 'react';
import AsyncStorage from '@react-native-async-storage/async-storage';

const FeatureFlagsContext = createContext();

const flagsReducer = (state, action) => {
  switch (action.type) {
    case 'LOAD_FLAGS':
      return { ...state, ...action.payload };
      
    case 'UPDATE_FLAG':
      return {
        ...state,
        [action.flagName]: {
          ...state[action.flagName],
          ...action.updates
        }
      };
      
    case 'ENABLE_FLAG':
      return {
        ...state,
        [action.flagName]: {
          ...state[action.flagName],
          enabled: true,
          lastModified: Date.now()
        }
      };
      
    case 'DISABLE_FLAG':
      return {
        ...state,
        [action.flagName]: {
          ...state[action.flagName],
          enabled: false,
          lastModified: Date.now()
        }
      };
      
    default:
      return state;
  }
};

export const FeatureFlagsProvider = ({ children }) => {
  const [flags, dispatch] = useReducer(flagsReducer, DEFAULT_FLAGS);
  
  // Cargar flags al inicio
  useEffect(() => {
    loadFlags();
  }, []);
  
  const loadFlags = async () => {
    try {
      // 1. Cargar flags locales
      const localFlags = await AsyncStorage.getItem('feature_flags');
      if (localFlags) {
        dispatch({ type: 'LOAD_FLAGS', payload: JSON.parse(localFlags) });
      }
      
      // 2. Fetch flags remotos (cuando tengamos backend)
      // const remoteFlags = await fetchRemoteFlags();
      // if (remoteFlags) {
      //   dispatch({ type: 'LOAD_FLAGS', payload: remoteFlags });
      //   await AsyncStorage.setItem('feature_flags', JSON.stringify(remoteFlags));
      // }
    } catch (error) {
      console.error('Error loading feature flags:', error);
    }
  };
  
  return (
    <FeatureFlagsContext.Provider value={{ flags, dispatch }}>
      {children}
    </FeatureFlagsContext.Provider>
  );
};
```

### Hook Principal: useFeatureFlag
```javascript
// hooks/useFeatureFlag.js
export const useFeatureFlag = (flagName) => {
  const { flags } = useContext(FeatureFlagsContext);
  
  const flag = flags[flagName];
  
  if (!flag) {
    console.warn(`Feature flag '${flagName}' not found`);
    return {
      enabled: false,
      loading: false,
      error: `Flag '${flagName}' not configured`
    };
  }
  
  // Evaluar condiciones
  const enabled = evaluateFlag(flag);
  
  return {
    enabled,
    loading: false,
    rollout: flag.rollout,
    metadata: flag.metadata
  };
};

// Evaluación de condiciones
const evaluateFlag = (flag) => {
  if (!flag.enabled) return false;
  
  // Rollout percentage
  if (flag.rollout < 100) {
    const userId = getUserId(); // hash del device
    const userBucket = hashCode(userId) % 100;
    if (userBucket >= flag.rollout) return false;
  }
  
  // Condiciones adicionales
  if (flag.conditions) {
    // Version check
    if (flag.conditions.minAppVersion) {
      const currentVersion = getAppVersion();
      if (!isVersionGreaterOrEqual(currentVersion, flag.conditions.minAppVersion)) {
        return false;
      }
    }
    
    // Platform check
    if (flag.conditions.platforms) {
      const currentPlatform = Platform.OS;
      if (!flag.conditions.platforms.includes(currentPlatform)) {
        return false;
      }
    }
  }
  
  return true;
};
```

### Hook de múltiples flags: useFeatureFlags
```javascript
// hooks/useFeatureFlags.js
export const useFeatureFlags = (flagNames) => {
  const { flags } = useContext(FeatureFlagsContext);
  
  const results = useMemo(() => {
    return flagNames.reduce((acc, flagName) => {
      const flag = flags[flagName];
      acc[flagName] = flag ? evaluateFlag(flag) : false;
      return acc;
    }, {});
  }, [flags, flagNames]);
  
  return results;
};

// Uso en componentes
const MyComponent = () => {
  const { auth_system, basic_stats, new_onboarding } = useFeatureFlags([
    FEATURE_FLAGS.AUTH_SYSTEM,
    FEATURE_FLAGS.BASIC_STATS,
    FEATURE_FLAGS.NEW_ONBOARDING
  ]);
  
  return (
    <View>
      {auth_system && <AuthButton />}
      {basic_stats && <StatsButton />}
      {new_onboarding ? <NewOnboarding /> : <ClassicOnboarding />}
    </View>
  );
};
```

```javascript
// components/FeatureFlagsDebugPanel.js
export const FeatureFlagsDebugPanel = ({ visible, onClose }) => {
  const { flags, dispatch } = useContext(FeatureFlagsContext);
  const [searchTerm, setSearchTerm] = useState('');
  
  const filteredFlags = Object.entries(flags).filter(([name]) =>
    name.toLowerCase().includes(searchTerm.toLowerCase())
  );
  
  const toggleFlag = (flagName) => {
    dispatch({
      type: flags[flagName].enabled ? 'DISABLE_FLAG' : 'ENABLE_FLAG',
      flagName
    });
  };
  
  return (
    <Modal visible={visible} animationType="slide">
      <SafeAreaView style={styles.container}>
        <View style={styles.header}>
          <Text style={styles.title}>Feature Flags Debug</Text>
          <TouchableOpacity onPress={onClose}>
            <Text style={styles.closeButton}>✕</Text>
          </TouchableOpacity>
        </View>
        
        <TextInput
          style={styles.searchInput}
          placeholder="Buscar flag..."
          value={searchTerm}
          onChangeText={setSearchTerm}
        />
        
        <ScrollView style={styles.flagsList}>
          {filteredFlags.map(([flagName, flag]) => (
            <FlagItem
              key={flagName}
              name={flagName}
              flag={flag}
              onToggle={() => toggleFlag(flagName)}
            />
          ))}
        </ScrollView>
      </SafeAreaView>
    </Modal>
  );
};

const FlagItem = ({ name, flag, onToggle }) => {
  const enabled = evaluateFlag(flag);
  
  return (

        <Text style={styles.flagDescription}>

        </Text>
```

## 📊 ANALYTICS DE FEATURE FLAGS

### Tracking de Usage
```javascript
// services/flagsAnalytics.js
class FlagsAnalytics {
  constructor() {
    this.events = [];
  }
  
  trackFlagEvaluation(flagName, enabled, conditions) {
    const event = {
      type: 'flag_evaluation',
      flagName,
      enabled,
      conditions,
      timestamp: Date.now(),
      userId: getUserId(),
      platform: Platform.OS,
      appVersion: getAppVersion()
    };
    
    this.events.push(event);
    
    // Enviar a analytics (batch)
    if (this.events.length >= 10) {
      this.flush();
    }
  }
  
  trackFlagInteraction(flagName, action, context) {
    const event = {
      type: 'flag_interaction',
      flagName,
      action, // 'enabled', 'disabled', 'viewed', etc.
      context,
      timestamp: Date.now(),
      userId: getUserId()
    };
    
    this.events.push(event);
  }
  
  async flush() {
    if (this.events.length === 0) return;
    
    try {
      // Enviar a backend cuando esté disponible
      // await sendAnalyticsEvents(this.events);
      
      // Por ahora, guardar local para debug
      await AsyncStorage.setItem(
        'flags_analytics', 
        JSON.stringify(this.events)
      );
      
      this.events = [];
    } catch (error) {
      console.error('Error sending flags analytics:', error);
    }
  }
}

export const flagsAnalytics = new FlagsAnalytics();
```

## 🔄 A/B TESTING

### Configuración de A/B Tests
```javascript
// A/B Testing configuration
const AB_TESTS = {
  onboarding_flow: {
    variants: {
      control: { weight: 50 }, // Onboarding actual
      variant_a: { weight: 25 }, // Onboarding simplificado
      variant_b: { weight: 25 }  // Onboarding con video
    },
    trafficAllocation: 30, // 30% de usuarios en test
    startDate: '2025-01-20',
    endDate: '2025-02-20',
    metrics: ['completion_rate', 'time_to_first_game', 'retention_d7']
  }
};

export const useABTest = (testName) => {
  const userId = getUserId();
  const [variant, setVariant] = useState(null);
  
  useEffect(() => {
    const test = AB_TESTS[testName];
    if (!test) return;
    
    // Determinar si el usuario está en el test
    const userBucket = hashCode(userId) % 100;
    if (userBucket >= test.trafficAllocation) {
      setVariant('not_in_test');
      return;
    }
    
    // Asignar variante basado en pesos
    const assignedVariant = assignVariant(userId, test.variants);
    setVariant(assignedVariant);
    
    // Track assignment
    flagsAnalytics.trackFlagInteraction(testName, 'assigned', {
      variant: assignedVariant,
      testConfig: test
    });
  }, [testName, userId]);
  
  return {
    variant,
    isInTest: variant && variant !== 'not_in_test',
    trackEvent: (event, data) => {
      flagsAnalytics.trackFlagInteraction(testName, event, {
        variant,
        eventData: data
      });
    }
  };
};
```
